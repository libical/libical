145static void buf_trim(struct buf *buf)
146{
147 /* trim whitespace
148 *
149 * XXX we can do this within the existing buffer
150 * because the new string length <= old string length
151 */
152 const char *s = buf_cstring(buf);
153
154 for (buf->len = 0; *s; s++) {
155 if (isspace((int)*s)) {
156 continue;
157 }
158 buf->s[buf->len++] = *s;
159 }
160}

163#pragma GCC diagnostic ignored "-Wformat-nonliteral"
164static void buf_vprintf(struct buf *buf, const char *fmt, va_list args)
165{
166 va_list ap;
167 size_t size, n;
168
169 /* Copy args in case we have to try again */
170 va_copy(ap, args);
171
172 size = buf->alloc - buf->len;
173 n = (size_t)vsnprintf(buf->s + buf->len, size, fmt, args);
174
175 if (n >= size) {
176 /* Grow the buffer and try again */
177 size = n + BUF_GROW;
178 buf->alloc += size;
179 buf->s = icalmemory_resize_buffer(buf->s, buf->alloc);
180
181 n = (size_t)vsnprintf(buf->s + buf->len, size, fmt, ap);
182 }
183 va_end(ap);
184
185 buf->len += n;
186}
187#pragma GCC diagnostic pop

189#define NOTESTART() state->itemstart = state->p
190#define MAKE(X, Y) X = icalmemory_new_buffer(sizeof(struct Y))
191#define PUTC(C) buf_putc(&state->buf, C)
192#define INC(I) state->p += I
193#define IS_CTRL(ch) \
194 ((ch) > 0 && (ch) <= 0x1f && (ch) != '\r' && (ch) != '\n' && (ch) != '\t')
195#define HANDLECTRL(state) \
196 { \
197 if (IS_CTRL(*(state)->p)) { \
198 while (IS_CTRL(*(state)->p)) \
199 (state)->p++; \
200 } \
201 if ((*(state)->p) == 0) \
202 break; \
203 }

205static int _parse_param_name(struct vcardparser_state *state)
206{
207 vcardparameter_kind kind;
208 const char *name;
209
210 while (*state->p) {
211 /* Handle control characters and break for NUL char */
212 HANDLECTRL(state);
213
214 switch (*state->p) {
215 case '=':
216 name = buf_cstring(&state->buf);
217 kind = vcardparameter_string_to_kind(name);
218 state->param = vcardparameter_new(kind);
219 if (kind == VCARD_X_PARAMETER) {
220 vcardparameter_set_xname(state->param, name);
221 } else if (kind == VCARD_IANA_PARAMETER) {
222 vcardparameter_set_iana_name(state->param, name);
223 }
224 buf_reset(&state->buf);
225 INC(1);
226 return 0;
227
228 case ';': /* vCard 2.1 parameter with just a value */
229 case ':':
230 state->param = vcardparameter_new(VCARD_TYPE_PARAMETER);
231 /* no INC - we need to see this char up a layer */
232 return 0;
233
234 case '\r':
235 INC(1);
236 break; /* just skip */
237 case '\n':
238 if (state->p[1] != ' ' && state->p[1] != '\t') {
239 return PE_KEY_EOL;
240 }
241 INC(2);
242 break;
243
244 /* XXX - check exact legal set? */
245 default:
246 PUTC(*state->p);
247 INC(1);
248 break;
249 }
250 }
251
252 return PE_KEY_EOF;
253}

255/* just leaves it on the buffer */
256static int _parse_param_quoted(struct vcardparser_state *state,
257 bool is_structured, bool is_multivalued)
258{
259 NOTESTART();
260
261 while (*state->p) {
262 /* Handle control characters and break for NUL char */
263 HANDLECTRL(state);
264
265 switch (*state->p) {
266 case '"':
267 INC(1);
268 return 0;
269
270 /* normal backslash quoting - NOTE, not strictly RFC compliant,
271 * but I figure anyone who generates one PROBABLY meant to escape
272 * the next character because it's so common, and LABEL definitely
273 * allows \n, so we have to handle that anyway */
274 case '\\':
275 /* seen in the wild - \n split by line wrapping */
276 if (state->p[1] == '\r') {
277 INC(1);
278 }
279 if (state->p[1] == '\n') {
280 if (state->p[2] != ' ' && state->p[2] != '\t') {
281 return PE_QSTRING_EOL;
282 }
283 INC(2);
284 }
285 if (!state->p[1]) {
286 return PE_BACKQUOTE_EOF;
287 }
288 if (state->p[1] == 'n' || state->p[1] == 'N') {
289 PUTC('\n');
290 } else if (is_structured && strchr(";,", state->p[1])) {
291 // preserve escaped COMMA and SEMICOLON in structured value
292 PUTC(state->p[0]);
293 PUTC(state->p[1]);
294 } else {
295 PUTC(state->p[1]);
296 }
297 INC(2);
298 break;
299
300 /* special value quoting for doublequote and endline (RFC 6868) */
301 case '^':
302 if (state->p[1] == '\r') {
303 INC(1);
304 }
305 if (state->p[1] == '\n') {
306 if (state->p[2] != ' ' && state->p[2] != '\t') {
307 return PE_QSTRING_EOL;
308 }
309 INC(2);
310 }
311 if (state->p[1] == '\'') {
312 PUTC('"');
313 INC(2);
314 } else if (state->p[1] == 'n') { /* only lower case per the RFC */
315 PUTC('\n');
316 INC(2);
317 } else if (state->p[1] == '^') {
318 PUTC('^');
319 INC(2);
320 } else {
321 PUTC('^');
322 INC(1); /* treat next char normally */
323 }
324 break;
325
326 case '\r':
327 INC(1);
328 break; /* just skip */
329 case '\n':
330 if (state->p[1] != ' ' && state->p[1] != '\t') {
331 return PE_QSTRING_EOL;
332 }
333 INC(2);
334 break;
335
336 case ',':
337 if (is_multivalued && !is_structured) {
338 return PE_QSTRING_EOV;
339 }
340 /* or fall through, comma isn't special */
341 _fallthrough();
342
343 default:
344 PUTC(*state->p);
345 INC(1);
346 break;
347 }
348 }
349
350 return PE_QSTRING_EOF;
351}

353static int _parse_param_value(struct vcardparser_state *state)
354{
355 bool is_multivalued = vcardparameter_is_multivalued(state->param);
356 bool is_structured = vcardparameter_is_structured(state->param);
357 int r;
358
359 while (*state->p) {
360 /* Handle control characters and break for NUL char */
361 HANDLECTRL(state);
362
363 switch (*state->p) {
364 case '\\': /* normal backslash quoting */
365 /* seen in the wild - \n split by line wrapping */
366 if (state->p[1] == '\r') {
367 INC(1);
368 }
369 if (state->p[1] == '\n') {
370 if (state->p[2] != ' ' && state->p[2] != '\t') {
371 return PE_PARAMVALUE_EOL;
372 }
373 INC(2);
374 }
375 if (!state->p[1]) {
376 return PE_BACKQUOTE_EOF;
377 }
378 if (state->p[1] == 'n' || state->p[1] == 'N') {
379 PUTC('\n');
380 } else {
381 PUTC(state->p[1]);
382 }
383 INC(2);
384 break;
385
386 case '^': /* special value quoting for doublequote (RFC 6868) */
387 /* seen in the wild - \n split by line wrapping */
388 if (state->p[1] == '\r') {
389 INC(1);
390 }
391 if (state->p[1] == '\n') {
392 if (state->p[2] != ' ' && state->p[2] != '\t') {
393 return PE_PARAMVALUE_EOL;
394 }
395 INC(2);
396 }
397 if (state->p[1] == '\'') {
398 PUTC('"');
399 INC(2);
400 } else if (state->p[1] == 'n') {
401 PUTC('\n');
402 INC(2);
403 } else if (state->p[1] == '^') {
404 PUTC('^');
405 INC(2);
406 } else {
407 PUTC('^');
408 INC(1); /* treat next char normally */
409 }
410 break;
411
412 case '"':
413 INC(1);
414 while ((r = _parse_param_quoted(state,
415 is_structured,
416 is_multivalued)) == PE_QSTRING_EOV) {
417 vcardparameter_add_value_from_string(state->param,
418 buf_cstring(&state->buf));
419
420 buf_reset(&state->buf);
421 INC(1);
422 }
423 if (r) {
424 return r;
425 }
426 break;
427
428 case ':':
429 case ';':
430 /* done - end of parameter */
431 if (is_multivalued) {
432 vcardparameter_add_value_from_string(state->param,
433 buf_cstring(&state->buf));
434 } else {
435 vcardparameter_set_value_from_string(state->param,
436 buf_cstring(&state->buf));
437
438 /* if it is a VALUE parameter, set the value kind */
439 if (vcardparameter_isa(state->param) == VCARD_VALUE_PARAMETER) {
440 vcardvalue_kind kind =
441 vcardvalue_string_to_kind(buf_cstring(&state->buf));
442 if (kind != VCARD_NO_VALUE) {
443 state->value_kind = kind;
444 }
445 }
446 }
447
448 buf_reset(&state->buf);
449 /* no INC - we need to see this char up a layer */
450 return 0;
451
452 case '\r':
453 INC(1);
454 break; /* just skip */
455
456 case '\n':
457 if (state->p[1] != ' ' && state->p[1] != '\t') {
458 return PE_PARAMVALUE_EOL;
459 }
460 INC(2);
461 break;
462
463 case ',':
464 if (is_multivalued) {
465 vcardparameter_add_value_from_string(state->param,
466 buf_cstring(&state->buf));
467 buf_reset(&state->buf);
468 INC(1);
469 break;
470 }
471 /* or fall through, comma isn't special */
472 _fallthrough();
473
474 default:
475 PUTC(*state->p);
476 INC(1);
477 break;
478 }
479 }
480
481 return PE_PARAMVALUE_EOF;
482}

484static void _parse_error(struct vcardparser_state *state,
485 enum vcardparameter_xlicerrortype type,
486 const char *fmt, ...)
487{

489

492 buf_vprintf(&state->errbuf, fmt, ap);
493 va_end(ap);
494
495 if (state->prop) {
496 vcardproperty_free(state->prop);

498 state->version = NULL;

503 errParam,
504 (void *)0);
505 buf_reset(&state->buf);

508static int _parse_prop_params(struct vcardparser_state *state)
509{
510 vcardproperty_kind prop_kind = vcardproperty_isa(state->prop);
511 const char *group = vcardproperty_get_group(state->prop);
512
513 do {
514 int r;
515
516 INC(1); /* skip ';' */
517
518 NOTESTART();
519
520 /* get the name */
521 r = _parse_param_name(state);
522 if (r) {
523 _parse_error(state,
524 VCARD_XLICERRORTYPE_PARAMETERNAMEPARSEERROR,
525 "%s '%s' in %s%s%s property. Removing entire property",
526 vcardparser_errstr(r), buf_cstring(&state->buf),
527 group ? group : "", group ? "." : "",
528 vcardproperty_kind_to_string(prop_kind));
529 return r;
530 }
531
532 vcardproperty_add_parameter(state->prop, state->param);
533
534 /* now get the value */
535 r = _parse_param_value(state);
536 if (r) {
537 vcardparameter_kind param_kind = vcardparameter_isa(state->param);
538
539 _parse_error(state,
540 VCARD_XLICERRORTYPE_PARAMETERVALUEPARSEERROR,
541 "%s for %s in %s%s%s property. Removing entire property",
542 vcardparser_errstr(r),
543 vcardparameter_kind_to_string(param_kind),
544 group ? group : "", group ? "." : "",
545 vcardproperty_kind_to_string(prop_kind));
546 return r;
547 }
548
549 } while (*state->p == ';'); /* another parameter to parse */
550
551 return 0;
552}

554static int _parse_prop_name(struct vcardparser_state *state)
555{
556 const char *name;
557 const char *group = NULL;
558 vcardproperty_kind kind;
559 vcardproperty_version version = VCARD_VERSION_NONE;
560 int r = 0;
561
562 NOTESTART();
563
564 size_t cnt = 0;
565 const size_t max_chars = 100; // plenty long enough for any property name
566 while (*state->p && cnt++ < max_chars) {
567 /* Handle control characters and break for NUL char */
568 HANDLECTRL(state);
569
570 switch (*state->p) {
571 case ':':
572 case ';':
573 name = buf_cstring(&state->buf);
574 kind = vcardproperty_string_to_kind(name);
575
576 state->prop = vcardproperty_new(kind);
577 if (!state->prop) {
578 return PE_NAME_INVALID;
579 }
580
581 if (kind == VCARD_X_PROPERTY) {
582 vcardproperty_set_x_name(state->prop, name);
583 }
584
585 if (group) {
586 vcardproperty_set_group(state->prop, group);
587 }
588
589 if (state->version) {
590 version = vcardproperty_get_version(state->version);
591 }
592
593 /* set default value kind */
594 switch (kind) {
595 case VCARD_GEO_PROPERTY:
596 state->value_kind = version == VCARD_VERSION_40 ? VCARD_URI_VALUE : VCARD_GEO_VALUE;
597 break;
598
599 case VCARD_KEY_PROPERTY:
600 case VCARD_LOGO_PROPERTY:
601 case VCARD_PHOTO_PROPERTY:
602 case VCARD_SOUND_PROPERTY:
603 state->value_kind = version == VCARD_VERSION_40 ? VCARD_URI_VALUE : VCARD_TEXT_VALUE;
604 break;
605
606 case VCARD_TZ_PROPERTY:
607 state->value_kind = version == VCARD_VERSION_40 ? VCARD_TEXT_VALUE : VCARD_UTCOFFSET_VALUE;
608 break;
609
610 case VCARD_UID_PROPERTY:
611 state->value_kind = version == VCARD_VERSION_40 ? VCARD_URI_VALUE : VCARD_TEXT_VALUE;
612 break;
613
614 case VCARD_X_PROPERTY:
615 state->value_kind =
616 xprop_value_kind_func ? xprop_value_kind_func(name, xprop_value_kind_data)
617 : VCARD_X_VALUE;
618 break;
619
620 default:
621 state->value_kind = vcardproperty_kind_to_value_kind(kind);
622 break;
623 }
624
625 buf_reset(&state->buf);
626
627 /* no INC - we need to see this char up a layer */
628 return r;
629
630 case '.':
631 if (group) {
632 size_t tmpLen = strlen(group) + buf_len(&state->buf) + 2;
633 char *tmp = icalmemory_tmp_buffer(tmpLen);
634 snprintf(tmp, tmpLen - 1, "%s.%s", group, buf_cstring(&state->buf));
635 group = tmp;
636 r = PE_PROP_MULTIGROUP;
637 } else {
638 group = icalmemory_tmp_copy(buf_cstring(&state->buf));
639 }
640 buf_reset(&state->buf);
641 INC(1);
642 break;
643
644 case '\r':
645 INC(1);
646 break; /* just skip */
647 case '\n':
648 if (state->p[1] == ' ' || state->p[1] == '\t') { /* wrapped line */
649 INC(2);
650 } else if (!state->buf.len) {
651 /* no key yet? blank intermediate lines are OK */
652 INC(1);
653 } else {
654 return PE_NAME_EOL;
655 }
656 break;
657
658 default:
659 PUTC(*state->p);
660 INC(1);
661 break;
662 }
663 }
664
665 return PE_NAME_EOF;
666}

668static int _parse_prop_value(struct vcardparser_state *state)
669{
670 vcardproperty_kind prop_kind = vcardproperty_isa(state->prop);
671 vcardvalue *value = NULL;
672
673 NOTESTART();
674
675 while (*state->p) {
676 /* Handle control characters and break for NUL char */
677 HANDLECTRL(state);
678
679 switch (*state->p) {
680 case '\\':
681 /* seen in the wild - \n split by line wrapping */
682 while (state->p[1] && strchr("\r\n", state->p[1])) {
683 if (state->p[1] == '\r') {
684 INC(1);
685 }
686 if (state->p[1] == '\n') {
687 if (state->p[2] != ' ' && state->p[2] != '\t') {
688 /* ignore unescaped backslash at end of line */
689 INC(2);
690 goto out;
691 }
692 INC(2);
693 }
694 }
695 if (state->p[1]) {
696 /* preserve escape sequences */
697 PUTC('\\');
698 PUTC(state->p[1]);
699 INC(2);
700 } else {
701 /* ignore unescaped backslash at end of line */
702 INC(1);
703 goto out;
704 }
705 break;
706 case '\r':
707 INC(1);
708 break; /* just skip */
709 case '\n':
710 if (state->p[1] == ' ' || state->p[1] == '\t') { /* wrapped line */
711 INC(2);
712 break;
713 }
714 /* otherwise it's the end of the value */
715 INC(1);
716 goto out;
717
718 default:
719 PUTC(*state->p);
720 INC(1);
721 break;
722 }
723 }
724
725out:
726 /* reaching the end of the file isn't a failure here,
727 * it's just another type of end-of-value */
728
729 /* repair critical property values */
730 if (prop_kind == VCARD_VERSION_PROPERTY) {
731 buf_trim(&state->buf);
732 state->version = state->prop;
733 }
734
735 if (state->value_kind == VCARD_TEXTLIST_VALUE) {
736 char sep = vcardproperty_is_structured(prop_kind) ? ';' : ','; //NOLINT(readability-implicit-bool-conversion)
737 vcardstrarray *textlist =
738 vcardtextlist_new_from_string(buf_cstring(&state->buf), sep);
739 if (textlist) {
740 value = vcardvalue_new_textlist(textlist);
741 }
742 } else {
743 value = vcardvalue_new_from_string(state->value_kind,
744 buf_cstring(&state->buf));
745 }
746
747 if (!value) {
748 return PE_VALUE_INVALID;
749 }
750
751 vcardproperty_set_value(state->prop, value);
752 buf_reset(&state->buf);
753
754 return 0;
755}

757static void _parse_eatline(struct vcardparser_state *state)
758{
759 while (*state->p) {
760 /* Handle control characters and break for NUL char */
761 HANDLECTRL(state);
762
763 switch (*state->p) {
764 case '\n':
765 if (state->p[1] == ' ' || state->p[1] == '\t') { /* wrapped line */
766 INC(2);
767 break;
768 }
769 /* otherwise it's the end of the line */
770 INC(1);
771 return;

773 default:
774 INC(1);
775 break;
776 }

780static void _parse_prop(struct vcardparser_state *state)
781{
782 int r = _parse_prop_name(state);
783 if (r) {
784 if (r == PE_PROP_MULTIGROUP) {
785 vcardproperty_kind prop_kind = vcardproperty_isa(state->prop);
786
787 _parse_error(state,
788 VCARD_XLICERRORTYPE_PROPERTYPARSEERROR,
789 "%s '%s.%s'. Removing entire property",
790 vcardparser_errstr(r),
791 vcardproperty_get_group(state->prop),
792 vcardproperty_kind_to_string(prop_kind));
793 _parse_eatline(state);
794 } else if (r == PE_NAME_INVALID) {
795 _parse_error(state,
796 VCARD_XLICERRORTYPE_PROPERTYPARSEERROR,
797 "%s '%s'. Removing entire property",
798 vcardparser_errstr(r), buf_cstring(&state->buf));
799 _parse_eatline(state);
800 } else {
801 _parse_error(state,
802 VCARD_XLICERRORTYPE_PROPERTYPARSEERROR,
803 "%s '%s'. Ignoring property",
804 vcardparser_errstr(r), buf_cstring(&state->buf));
805 }
806 return;
807 }
808
809 if (*state->p == ';') {
810 r = _parse_prop_params(state);
811 if (r) {
812 /* errors handled in _parse_prop_params() */
813 return;
814 }
815 }
816
817 INC(1); /* skip ':' */
818 r = _parse_prop_value(state);
819 if (r) {
820 vcardproperty_kind prop_kind = vcardproperty_isa(state->prop);
821 const char *group = vcardproperty_get_group(state->prop);
822
823 if (r == PE_VALUE_INVALID) {
824 _parse_error(state,
825 VCARD_XLICERRORTYPE_VALUEPARSEERROR,
826 "Error parsing '%s' as %s value in %s%s%s property."
827 " Removing entire property",
828 buf_cstring(&state->buf),
829 vcardvalue_kind_to_string(state->value_kind),
830 group ? group : "", group ? "." : "",
831 vcardproperty_kind_to_string(prop_kind));
832 } else {
833 _parse_error(state,
834 VCARD_XLICERRORTYPE_VALUEPARSEERROR,
835 "%s in %s%s%s property. Removing entire property",
836 vcardparser_errstr(r),
837 group ? group : "", group ? "." : "",
838 vcardproperty_kind_to_string(prop_kind));
839 }

843static int _parse_vcard(struct vcardparser_state *state, int only_one)
844{
845 const char *cardstart = state->p;
846 int r = 0;
847
848 while (*state->p) {
849 /* whitespace is very skippable before AND afterwards */
850 if (*state->p == '\r' || *state->p == '\n' ||
851 *state->p == ' ' || *state->p == '\t') {
852 INC(1);
853 continue;
854 }
855
856 _parse_prop(state);
857
858 if (vcardproperty_isa(state->prop) == VCARD_BEGIN_PROPERTY) {
859 if (state->vcard) {
860 // Reject nested VCARD.
861 state->itemstart = cardstart;
862 r = PE_MISMATCHED_CARD;
863 break;
864 }
865
866 if (vcardvalue_isa(vcardproperty_get_value(state->prop)) !=
867 VCARD_TEXT_VALUE) {
868 r = PE_VALUE_INVALID;
869 break;
870 }
871
872 const char *val =
873 vcardvalue_get_text(vcardproperty_get_value(state->prop));
874
875 if (!val || strcasecmp(val, "VCARD") != 0) {
876 state->itemstart = cardstart;
877 r = PE_NAME_INVALID;
878 break;
879 }
880
881 vcardproperty_free(state->prop);
882 state->prop = NULL;
883 state->vcard = vcardcomponent_new(VCARD_VCARD_COMPONENT);
884 vcardcomponent_add_component(state->xroot, state->vcard);
885
886 } else if (!state->vcard) {
887 /* this means we're at the top level, haven't seen a BEGIN! */
888 state->itemstart = cardstart;
889 r = PE_MISMATCHED_CARD;
890 break;
891 } else if (vcardproperty_isa(state->prop) == VCARD_END_PROPERTY) {
892 if (vcardvalue_isa(vcardproperty_get_value(state->prop)) !=
893 VCARD_TEXT_VALUE) {
894 r = PE_VALUE_INVALID;
895 break;
896 }
897
898 const char *val =
899 vcardvalue_get_text(vcardproperty_get_value(state->prop));
900
901 if (!val || strcasecmp(val, "VCARD") != 0) {
902 state->itemstart = cardstart;
903 r = PE_NAME_INVALID;
904 break;
905 }
906
907 vcardproperty_free(state->prop);
908 state->vcard = NULL;
909 state->prop = NULL;
910
911 if (only_one) {
912 break;
913 }
914
915 } else {
916 vcardcomponent_add_property(state->vcard, state->prop);
917 state->prop = NULL;
918 }
919 }
920
921 if (state->prop) {
922 if (vcardproperty_isa(state->prop) != VCARD_END_PROPERTY) {
923 r = PE_FINISHED_EARLY;
924 }
925
926 vcardproperty_free(state->prop);
927 state->prop = NULL;
928 }
929
930 return r;
931}

933static int vcardparser_parse(struct vcardparser_state *state, int only_one)
934{
935 state->xroot = vcardcomponent_new(VCARD_XROOT_COMPONENT);
936
937 state->p = state->base;
938
939 buf_init(&state->buf, BUF_GROW);
940 buf_init(&state->errbuf, BUF_GROW);
941
942 /* don't parse trailing non-whitespace */
943 return _parse_vcard(state, only_one);
944}

1010vcardcomponent *vcardparser_parse_string(const char *str)
1011{
1012 struct vcardparser_state parser;
1013 vcardcomponent *vcard = NULL;
1014 int r;
1015
1016 memset(&parser, 0, sizeof(struct vcardparser_state));
1017
1018 parser.base = str;
1019 r = vcardparser_parse(&parser, 0);
1020 if (!r) {
1021 if (vcardcomponent_count_components(parser.xroot,
1022 VCARD_VCARD_COMPONENT) == 1) {
1023 vcard = vcardcomponent_get_first_component(parser.xroot,
1024 VCARD_VCARD_COMPONENT);
1025 vcardcomponent_remove_component(parser.xroot, vcard);
1026 } else {
1027 vcard = parser.xroot;
1028 parser.xroot = NULL;
1029 }
1030 }
1031
1032 vcardparser_free(&parser);
1033
1034 return vcard;
1035}
